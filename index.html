/* ---------------- Profile photos (Drive folder "Staff Photos") ---------------- */ const TD_PHOTO_FOLDER = 'Staff Photos'; const TD_PHOTO_MAX = 2 * 1024 * 1024; // 2 MB after browser resize function TD_photoFolder_() { const props = PropertiesService.getScriptProperties(); const id = props.getProperty('TD_PHOTO_FOLDER_ID'); if (id) { try { return DriveApp.getFolderById(id); } catch (e) {} } const it = DriveApp.getRootFolder().getFoldersByName(TD_PHOTO_FOLDER); const folder = it.hasNext() ? it.next() : DriveApp.getRootFolder().createFolder(TD_PHOTO_FOLDER); try { folder.setSharing(DriveApp.Access.ANYONE, DriveApp.Permission.VIEW); } catch (e) { folder.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW); } props.setProperty('TD_PHOTO_FOLDER_ID', folder.getId()); return folder; } function TD_safeId_(userId) { return String(userId).toLowerCase().replace(/[^a-z0-9._-]/g, '_'); } // File name = _.jpg e.g. gagan_4332.jpg function TD_photoName_(userId, pass) { const p = String(pass || '').trim().replace(/[^A-Za-z0-9._-]/g, '_'); return TD_safeId_(userId) + (p ? '_' + p : '') + '.jpg'; } // Password of a user from the Users tab (col I) function TD_passOf_(userId) { const u = TD_users_().find(r => String(r[TD_U.USER]).trim().toLowerCase() === String(userId).toLowerCase()); return u ? String(u[TD_U.PASS]).trim() : ''; } // All photos of this user: ".jpg" or "_.jpg" function TD_userPhotos_(folder, userId) { const id = TD_safeId_(userId), out = []; const it = folder.searchFiles("title contains '" + id + "' and trashed = false"); while (it.hasNext()) { const f = it.next(), t = f.getName().toLowerCase(); if (t === id + '.jpg' || t.indexOf(id + '_') === 0) out.push(f); } return out; } function TD_photoUrl_(userId, pass) { try { const want = TD_photoName_(userId, pass); const files = TD_userPhotos_(TD_photoFolder_(), userId); if (!files.length) return ''; const f = files.find(x => x.getName() === want) || files[0]; if (f.getName() !== want) f.setName(want); // old name / password changed -> rename return 'https://drive.google.com/thumbnail?id=' + f.getId() + '&sz=w300&v=' + f.getLastUpdated().getTime(); } catch (e) { return ''; } } function TD_uploadPhoto_(token, data, mime) { const s = TD_session_(token); if (!s) return { ok: false, error: 'SESSION_EXPIRED' }; if (!data) return { ok: false, error: 'No photo received' }; if (!/^image\/(jpeg|png|webp)$/.test(String(mime))) return { ok: false, error: 'Only JPG / PNG photos allowed' }; const bytes = Utilities.base64Decode(String(data).replace(/^data:[^,]+,/, '')); if (bytes.length > TD_PHOTO_MAX) return { ok: false, error: 'Photo too large (max 2 MB)' }; const folder = TD_photoFolder_(); TD_userPhotos_(folder, s.userId).forEach(f => f.setTrashed(true)); // remove this user's old photo const name = TD_photoName_(s.userId, TD_passOf_(s.userId)); // e.g. gagan_4332.jpg const file = folder.createFile(Utilities.newBlob(bytes, 'image/jpeg', name)); try { file.setSharing(DriveApp.Access.ANYONE, DriveApp.Permission.VIEW); } catch (e) { file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW); } return { ok: true, photo: 'https://drive.google.com/thumbnail?id=' + file.getId() + '&sz=w300&v=' + Date.now() }; } // Run ONCE from the editor: approves Drive access and creates/shares the "Staff Photos" folder. function TD_setupPhotos() { const f = TD_photoFolder_(); Logger.log('Staff Photos folder: %s', f.getUrl()); }
